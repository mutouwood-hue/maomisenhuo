# ESP-IDF 5.5.5 collects vendor CMake requirements in a separate CMake process.
# Preserve that implementation, then append the manifest edges locally using
# the SDK property API. This hook changes no SDK/vendor source or checksum file.
# It is deliberately tied to the exact SDK; changing SDK requires re-review.
function(__component_get_requirements)
    idf_component_set_property(espressif__button COMPONENT_VERSION "4.2.2")
    idf_component_set_property(espressif__cmake_utilities COMPONENT_VERSION "0.5.3")
    idf_component_set_property(espressif__esp_codec_dev COMPONENT_VERSION "1.5.11")
    idf_component_set_property(espressif__esp_io_expander COMPONENT_VERSION "1.2.1")
    idf_component_set_property(espressif__esp_lcd_co5300 COMPONENT_VERSION "2.2.0")
    idf_component_set_property(espressif__esp_lcd_panel_io_additions COMPONENT_VERSION "1.0.1~1")
    idf_component_set_property(espressif__esp_lcd_touch COMPONENT_VERSION "1.2.1")
    idf_component_set_property(espressif__esp_lv_decoder COMPONENT_VERSION "0.4.3")
    idf_component_set_property(espressif__esp_lv_fs COMPONENT_VERSION "1.0.1")
    idf_component_set_property(espressif__esp_lvgl_adapter COMPONENT_VERSION "0.6.4")
    idf_component_set_property(espressif__esp_mmap_assets COMPONENT_VERSION "2.0.1")
    idf_component_set_property(espressif__esp_new_jpeg COMPONENT_VERSION "1.0.2")
    idf_component_set_property(espressif__freetype COMPONENT_VERSION "2.14.3~1")
    idf_component_set_property(espressif__knob COMPONENT_VERSION "1.1.0")
    idf_component_set_property(espressif__libpng COMPONENT_VERSION "1.6.58~2")
    idf_component_set_property(espressif__zlib COMPONENT_VERSION "1.3.2~1")
    idf_component_set_property(lvgl__lvgl COMPONENT_VERSION "9.5.0")
    idf_component_set_property(waveshare__esp32_s3_touch_amoled_1_75c COMPONENT_VERSION "3.0.0")
    idf_component_set_property(waveshare__esp_lcd_touch_cst9217 COMPONENT_VERSION "2.0.0")
    ___component_get_requirements()
    idf_component_set_property(espressif__esp_lcd_panel_io_additions REQUIRES "espressif__esp_io_expander" APPEND)
    idf_component_set_property(espressif__esp_lvgl_adapter REQUIRES "espressif__button;espressif__esp_lcd_touch;espressif__esp_lv_decoder;espressif__esp_lv_fs;espressif__freetype;espressif__knob" APPEND)
    idf_component_set_property(waveshare__esp32_s3_touch_amoled_1_75c REQUIRES "espressif__esp_codec_dev;espressif__esp_lvgl_adapter" APPEND)
    idf_component_set_property(waveshare__esp_lcd_touch_cst9217 REQUIRES "espressif__esp_lcd_touch" APPEND)
    idf_component_set_property(main REQUIRES "lvgl__lvgl" APPEND)
    idf_component_set_property(espressif__button PRIV_REQUIRES "espressif__cmake_utilities" APPEND)
    idf_component_set_property(espressif__esp_lcd_co5300 PRIV_REQUIRES "espressif__cmake_utilities" APPEND)
    idf_component_set_property(espressif__esp_lcd_panel_io_additions PRIV_REQUIRES "espressif__cmake_utilities" APPEND)
    idf_component_set_property(espressif__esp_lv_decoder PRIV_REQUIRES "espressif__esp_new_jpeg;espressif__libpng;lvgl__lvgl" APPEND)
    idf_component_set_property(espressif__esp_lv_fs PRIV_REQUIRES "espressif__cmake_utilities;espressif__esp_mmap_assets;lvgl__lvgl" APPEND)
    idf_component_set_property(espressif__esp_lvgl_adapter PRIV_REQUIRES "lvgl__lvgl" APPEND)
    idf_component_set_property(espressif__esp_mmap_assets PRIV_REQUIRES "espressif__cmake_utilities" APPEND)
    idf_component_set_property(espressif__knob PRIV_REQUIRES "espressif__cmake_utilities" APPEND)
    idf_component_set_property(espressif__libpng PRIV_REQUIRES "espressif__zlib" APPEND)
    idf_component_set_property(waveshare__esp32_s3_touch_amoled_1_75c PRIV_REQUIRES "espressif__esp_lcd_co5300;espressif__esp_lcd_panel_io_additions;lvgl__lvgl;waveshare__esp_lcd_touch_cst9217" APPEND)
    idf_component_set_property(main PRIV_REQUIRES "waveshare__esp32_s3_touch_amoled_1_75c" APPEND)
    # Qualify vendor short names using only this fixed local manifest graph.
    set(local2_managed_names
        espressif__button
        espressif__cmake_utilities
        espressif__esp_codec_dev
        espressif__esp_io_expander
        espressif__esp_lcd_co5300
        espressif__esp_lcd_panel_io_additions
        espressif__esp_lcd_touch
        espressif__esp_lv_decoder
        espressif__esp_lv_fs
        espressif__esp_lvgl_adapter
        espressif__esp_mmap_assets
        espressif__esp_new_jpeg
        espressif__freetype
        espressif__knob
        espressif__libpng
        espressif__zlib
        lvgl__lvgl
        waveshare__esp32_s3_touch_amoled_1_75c
        waveshare__esp_lcd_touch_cst9217
    )
    foreach(local2_owner IN ITEMS ${local2_managed_names} main)
        foreach(local2_property IN ITEMS REQUIRES PRIV_REQUIRES)
            idf_component_get_property(local2_edges ${local2_owner} ${local2_property})
            set(local2_qualified "")
            foreach(local2_edge IN LISTS local2_edges)
                foreach(local2_managed IN LISTS local2_managed_names)
                    string(REGEX REPLACE "^.*__" "" local2_short "${local2_managed}")
                    if(local2_edge STREQUAL local2_short)
                        set(local2_edge "${local2_managed}")
                    endif()
                endforeach()
                list(APPEND local2_qualified "${local2_edge}")
            endforeach()
            list(REMOVE_DUPLICATES local2_qualified)
            idf_component_set_property(${local2_owner} ${local2_property} "${local2_qualified}")
        endforeach()
    endforeach()

endfunction()
